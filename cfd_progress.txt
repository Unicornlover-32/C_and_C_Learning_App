#include <stdio.h>

int main(void)
{
    int score;
    scanf("%d", &score);

    if (score >= 90) {
        printf("Grade: A\n");
    } else if (score >= 80) {
        printf("Grade: B\n");
    } else if (score >= 70) {
        printf("Grade: C\n");
    } else if (score >= 60) {
        printf("Grade: D\n");
    } else {
        printf("Grade: F\n");
    }

    return 0;
}

#include <stdio.h>

int main(void)
{
    int n;
    scanf("%d", &n);

    int total = 0;
    for (int i = 1; i <= n; i++) {
        printf("%d\n", i);
        total += i;
    }
    printf("Total: %d\n", total);

    return 0;
}
